"""Local read-only real-state benchmark; invoke from the reviewed repository root.

Run with Python at the reviewed commit; see resource-report.md for the command.
Network access is limited to authenticated GitHub GETs through gh. Payload stays
in ignored .tmp; only bounded, sanitized evidence is emitted next to this file.
"""

import hashlib
import json
import os
import platform
import resource
import subprocess
import sys
import tempfile
import time
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path.cwd()
sys.path.insert(0, str(ROOT / "tools"))
sys.path.insert(0, str(ROOT / "src"))
import legislation_parent_state as parent
from archive_govt_nz.domains.legislation.cli_state import verify_linked_state

TARGET = "fcb8f7c89ffe18afd854f23f40c448fcc3a9efeb"
REPO = "edithatogo/archive-govt-nz"
OUT = Path(__file__).resolve().parent
COMMANDS = []
SOURCES = {}


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def save(name, value):
    raw = (json.dumps(value, indent=2, sort_keys=True) + "\n").encode()
    (OUT / name).write_bytes(raw)
    return sha(raw)


def command(args, timeout=120):
    COMMANDS.append(args)
    result = subprocess.run(args, capture_output=True, timeout=timeout, check=False)
    if result.returncode:
        raise RuntimeError(f"command failed ({result.returncode}); stderr withheld")
    return result.stdout


def api(path):
    raw = command(["gh", "api", f"repos/{REPO}/{path}"])
    SOURCES[path] = sha(raw)
    return json.loads(raw)


def snapshot(state):
    return {
        p.relative_to(state).as_posix(): sha(p.read_bytes())
        for p in sorted(state.rglob("*"))
        if p.is_file()
    }


def main():
    if not __debug__:
        raise RuntimeError("assertion checks require Python without optimization")
    if command(["git", "rev-parse", "HEAD"]).decode().strip() != TARGET:
        raise RuntimeError("run at the exact reviewed integration commit")
    if os.environ.get("GITHUB_ACTIONS") or os.environ.get("GITHUB_RUN_ID"):
        raise RuntimeError("this probe requires a local context")
    scratch_root = ROOT / ".tmp"
    scratch_root.mkdir(exist_ok=True)
    command(["git", "check-ignore", ".tmp/probe-state.zip"])
    scratch = Path(tempfile.mkdtemp(prefix="ordered-resource-", dir=scratch_root))
    reference_path = ROOT / "config/legislation/parents/ordered-500-33968609350.json"
    reference = json.loads(reference_path.read_bytes())
    artifact = api("actions/artifacts/9970365066")
    run = api("actions/runs/33968609350")
    parent.check_metadata(
        reference, {"artifact": artifact, "run": run}, datetime.now(UTC)
    )
    download_start = time.monotonic()
    raw = command(["gh", "api", f"repos/{REPO}/actions/artifacts/9970365066/zip"])
    download_seconds = time.monotonic() - download_start
    assert len(raw) == artifact["size_in_bytes"] == 10628850
    assert "sha256:" + sha(raw) == artifact["digest"] == reference["artifact"]["digest"]
    (scratch / "state.zip").write_bytes(raw)
    metadata_hash = save(
        "artifact-readback.json",
        {
            "artifact": {
                key: artifact[key] for key in (*reference["artifact"], "workflow_run")
            },
            "run": {
                key: run[key]
                for key in (
                    "id",
                    "head_sha",
                    "head_branch",
                    "run_attempt",
                    "workflow_id",
                    "name",
                    "path",
                    "status",
                    "conclusion",
                )
            },
            "source_sha256": dict(SOURCES),
            "metadata_verified": True,
            "download_sha256": sha(raw),
            "download_bytes": len(raw),
            "download_seconds_excluded_from_benchmark": download_seconds,
        },
    )
    start = time.monotonic()
    before = resource.getrusage(resource.RUSAGE_SELF)
    files = parent.unpack(raw)
    assert len(files) <= 1100 and sum(map(len, files.values())) <= 128 * 1024**2
    parent.verify_parent(files, reference)
    state = scratch / "state"
    for name, data in files.items():
        dest = state / name
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_bytes(data)
    original = snapshot(state)
    objects = verify_linked_state(
        state / "cas", state / "checkpoint.json", state / "manifest.json"
    )
    assert objects == 904
    reconciliation = scratch / "reconciliation.json"
    args = [
        sys.executable,
        "tools/reconcile_one_legislation_batch.py",
        "--batch-id",
        "prompt13-ordered500-20260905",
        "--batch-path",
        "seeds/reviewed/historical-work-ids-0001.txt",
        "--expected-batch-sha256",
        reference["source"]["seed_sha256"],
        "--manifest-path",
        str(state / "manifest.json"),
        "--checkpoint-path",
        str(state / "checkpoint.json"),
        "--cas-path",
        str(state / "cas"),
        "--receipt-path",
        str(reconciliation),
    ]
    command(args, timeout=60)
    result = json.loads(reconciliation.read_bytes())
    assert result["mismatch_count"] == 0 and result["reconciled_work_ids_count"] == 500
    assert snapshot(state) == original
    elapsed = time.monotonic() - start
    after = resource.getrusage(resource.RUSAGE_SELF)
    children = resource.getrusage(resource.RUSAGE_CHILDREN)
    rss_scale = 1 if sys.platform == "darwin" else 1024
    rss = after.ru_maxrss * rss_scale
    child_rss = children.ru_maxrss * rss_scale
    passed = elapsed <= 60 and rss <= 512 * 1024**2 and child_rss <= 512 * 1024**2
    reconciliation_hash = save("resource-reconciliation.json", result)
    save(
        "resource-receipt.json",
        {
            "schema_version": "archive-govt-nz.local-resource-assurance/v1",
            "target_commit": TARGET,
            "execution_context": "local",
            "status": "passed" if passed else "failed",
            "observed_at": datetime.now(UTC).isoformat(),
            "platform": platform.platform(),
            "python": sys.version,
            "uv": command(["uv", "--version"]).decode().strip(),
            "probe_sha256": sha(Path(__file__).read_bytes()),
            "uv_lock_sha256": sha((ROOT / "uv.lock").read_bytes()),
            "reference_sha256": sha(reference_path.read_bytes()),
            "artifact_readback_sha256": metadata_hash,
            "reconciliation_sha256": reconciliation_hash,
            "objects_verified": objects,
            "physical_state_files": len(files),
            "state_bytes": sum(map(len, files.values())),
            "state_unchanged": True,
            "state_file_hashes_sha256": sha(
                json.dumps(original, sort_keys=True).encode()
            ),
            "mismatch_count": 0,
            "scope_works": 500,
            "scope_records": result["selected_records_count"],
            "wall_seconds": elapsed,
            "user_seconds": after.ru_utime - before.ru_utime,
            "system_seconds": after.ru_stime - before.ru_stime,
            "peak_rss_bytes": rss,
            "child_peak_rss_bytes": child_rss,
            "bounds": {
                "wall_seconds": 60,
                "each_process_peak_rss_bytes": 512 * 1024**2,
                "expanded_bytes": 128 * 1024**2,
                "files": 1100,
            },
            "measurement_scope": "unpack, native parent verification, scratch materialization, linked CAS verification, reconciliation, before/after file hashing; network excluded",
            "rss_scope": "process lifetime high-water marks; children includes prior gh/git calls; separate maxima are not additive concurrent memory",
            "bounds_policy": "declared local acceptance budgets, not an inherited performance SLA; time/RSS assessed after execution, subprocesses timeout bounded",
            "source_acquisition_performed": False,
            "publication_performed": False,
            "commands": COMMANDS.copy(),
        },
    )
    assert passed, "resource budget exceeded; failed receipt retained"
    security = {}
    for kind in ("code-scanning", "dependabot", "secret-scanning"):
        endpoint = f"{kind}/alerts?state=open&per_page=100"
        raw_pages = command(["gh", "api", "--paginate", "--slurp", f"repos/{REPO}/{endpoint}"])
        SOURCES[endpoint] = sha(raw_pages)
        security[kind] = [
            {"number": row["number"], "state": row["state"]}
            for page in json.loads(raw_pages) for row in page
        ]
    ruleset = api("rulesets/22180861")
    rule_list = api("rulesets")
    rules = {rule["type"]: rule.get("parameters", {}) for rule in ruleset["rules"]}
    expected = {
        "Assurance (ubuntu-latest)",
        "Assurance (macos-latest)",
        "Assurance (windows-latest)",
        "analyze",
        "lint",
        "codecov/patch",
    }
    assert ruleset["enforcement"] == "active"
    assert "~DEFAULT_BRANCH" in ruleset["conditions"]["ref_name"]["include"]
    assert any(
        row["id"] == 22180861 and row["enforcement"] == "active" for row in rule_list
    )
    assert rules["required_status_checks"]["strict_required_status_checks_policy"]
    assert expected <= {
        x["context"] for x in rules["required_status_checks"]["required_status_checks"]
    }
    assert {"deletion", "non_fast_forward", "pull_request"} <= rules.keys()
    assert (
        rules["pull_request"]["require_review_thread_resolution"]
        if "require_review_thread_resolution" in rules["pull_request"]
        else rules["pull_request"]["required_review_thread_resolution"]
    )
    save(
        "security-ruleset-readback.json",
        {
            "target_commit": TARGET,
            "observed_at": datetime.now(UTC).isoformat(),
            "execution_context": "local_authenticated_read_only",
            "alerts": security,
            "status": "passed" if not any(security.values()) else "requires_review",
            "ruleset": {
                key: value
                for key, value in ruleset.items()
                if key not in ("node_id", "_links")
            },
            "source_sha256": SOURCES,
            "commands": COMMANDS,
            "probe_sha256": sha(Path(__file__).read_bytes()),
            "remote_mutations": False,
            "limitations": "current hosted policy/alert observation, not exact-head CI or a local dependency/secret scan; maintainer always-bypass is retained",
        },
    )
    save(
        "receipt-index.json",
        {
            p.name: sha(p.read_bytes())
            for p in sorted(OUT.iterdir())
            if p.suffix in (".txt", ".json") and p.name != "receipt-index.json"
        },
    )
    print(
        json.dumps(
            {
                "status": "passed",
                "seconds": elapsed,
                "objects": objects,
                "scratch": str(scratch),
            }
        )
    )


if __name__ == "__main__":
    main()
